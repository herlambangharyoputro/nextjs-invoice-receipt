-- ================================================
-- INVOICE & RECEIPT MANAGEMENT SYSTEM
-- Database Schema with Supabase Auth
-- ================================================

-- ================================================
-- 1. USER PROFILES (Extend Supabase Auth)
-- ================================================

CREATE TABLE user_profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name VARCHAR(255),
  avatar_url TEXT,
  phone VARCHAR(50),
  
  -- Role management
  role VARCHAR(50) DEFAULT 'user', -- 'admin', 'user', 'accountant', 'viewer'
  
  -- Profile relation
  profile_id UUID,
  
  -- Preferences
  language VARCHAR(10) DEFAULT 'id',
  timezone VARCHAR(50) DEFAULT 'Asia/Jakarta',
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ================================================
-- 2. PROFILES (Company Information)
-- ================================================

CREATE TABLE profiles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_name VARCHAR(255),
  tagline TEXT,
  email VARCHAR(255),
  phone VARCHAR(50),
  address TEXT,
  logo_url TEXT,
  
  -- Authorized signature
  authorized_person_name VARCHAR(255),
  authorized_person_title VARCHAR(255),
  signature_url TEXT,
  
  -- Email settings
  smtp_host VARCHAR(255),
  smtp_port INTEGER,
  smtp_user VARCHAR(255),
  smtp_password VARCHAR(255),
  email_from VARCHAR(255),
  email_signature TEXT,
  
  -- Audit
  created_by UUID REFERENCES auth.users(id),
  updated_by UUID REFERENCES auth.users(id),
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Add foreign key to user_profiles after profiles table created
ALTER TABLE user_profiles

  FOREIGN KEY (profile_id) REFERENCES profiles(id);

-- ================================================
-- 3. CLIENTS (Customer Data)
-- ================================================

CREATE TABLE clients (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255),
  phone VARCHAR(50),
  address TEXT,
  company_name VARCHAR(255),
  
  -- Email notification settings
  enable_email_notification BOOLEAN DEFAULT true,
  
  -- Audit
  created_by UUID REFERENCES auth.users(id),
  updated_by UUID REFERENCES auth.users(id),
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ================================================
-- 4. TAX SETTINGS
-- ================================================

CREATE TABLE tax_settings (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tax_name VARCHAR(100) NOT NULL, -- PPN, PPh, etc
  tax_rate DECIMAL(5,2) NOT NULL, -- 11.00 untuk PPN 11%
  is_active BOOLEAN DEFAULT true,
  
  created_by UUID REFERENCES auth.users(id),
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()

-- ================================================
-- 5. PAYMENT METHODS
-- ================================================

CREATE TABLE payment_methods (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  bank_name VARCHAR(100),
  account_holder VARCHAR(255),
  account_number VARCHAR(100),
  is_active BOOLEAN DEFAULT true,
  
  created_by UUID REFERENCES auth.users(id),
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ================================================
-- 6. INVOICES
-- ================================================

CREATE TABLE invoices (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  invoice_number VARCHAR(100) UNIQUE NOT NULL,
  client_id UUID REFERENCES clients(id),
  issue_date DATE NOT NULL,
  due_date DATE NOT NULL,
  subtotal DECIMAL(15,2) NOT NULL,
  
  -- Tax settings
  tax_id UUID REFERENCES tax_settings(id),
  tax_rate DECIMAL(5,2) DEFAULT 0, -- snapshot of tax rate when invoice created
  tax_amount DECIMAL(15,2) DEFAULT 0,
  show_tax BOOLEAN DEFAULT false, -- show/hide tax in invoice display
  
  -- Discount
  discount DECIMAL(15,2) DEFAULT 0,
  
  -- Total
  total DECIMAL(15,2) NOT NULL,
  
  -- Payment tracking
  payment_status VARCHAR(50) DEFAULT 'unpaid', -- 'unpaid', 'partial', 'paid', 'overdue'
  amount_paid DECIMAL(15,2) DEFAULT 0,
  amount_due DECIMAL(15,2),
  paid_at TIMESTAMPTZ,
  
  -- Signature (override from profile if needed)
  authorized_person_name VARCHAR(255),
  authorized_person_title VARCHAR(255),
  signature_url TEXT,
  
  -- Digital features
  public_url VARCHAR(255) UNIQUE,
  is_public BOOLEAN DEFAULT false,
  
  -- Email reminder settings
  send_reminder BOOLEAN DEFAULT true,
  reminder_sent_count INTEGER DEFAULT 0,
  last_reminder_sent_at TIMESTAMPTZ,
  
  -- Status
  status VARCHAR(50) DEFAULT 'draft', -- 'draft', 'sent', 'cancelled'
  notes TEXT,
  
  -- Audit
  created_by UUID REFERENCES auth.users(id),
  updated_by UUID REFERENCES auth.users(id),
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ================================================
-- 7. INVOICE ITEMS
-- ================================================

CREATE TABLE invoice_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  invoice_id UUID REFERENCES invoices(id) ON DELETE CASCADE,
  item_number INTEGER NOT NULL, -- sequence number (1, 2, 3, ...)
  item_description TEXT NOT NULL,
  quantity DECIMAL(10,2) NOT NULL, -- support decimal for units like hours, kg, etc
  unit VARCHAR(50) DEFAULT 'pcs', -- pcs, hours, kg, meter, etc
  unit_price DECIMAL(15,2) NOT NULL,
  subtotal DECIMAL(15,2) NOT NULL,
  
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ================================================
-- 8. RECEIPTS (Payment Receipts)
-- ================================================

CREATE TABLE receipts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  receipt_number VARCHAR(100) UNIQUE NOT NULL,
  invoice_id UUID REFERENCES invoices(id),
  payment_date DATE NOT NULL,
  amount_paid DECIMAL(15,2) NOT NULL,
  payment_method_id UUID REFERENCES payment_methods(id),
  
  -- Payment proof
  payment_proof_url TEXT, -- upload bukti transfer
  
  -- Signature for receipt
  authorized_person_name VARCHAR(255),
  authorized_person_title VARCHAR(255),
  signature_url TEXT,
  
  -- Digital features
  public_url VARCHAR(255) UNIQUE,
  is_public BOOLEAN DEFAULT false,
  
  notes TEXT,
  
  created_by UUID REFERENCES auth.users(id),
  
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- FUNCTIONS & TRIGGERS
-- ================================================

-- Function: Auto calculate amount_due and payment_status
CREATE OR REPLACE FUNCTION calculate_amount_due()
RETURNS TRIGGER AS $$
BEGIN
  NEW.amount_due = NEW.total - COALESCE(NEW.amount_paid, 0);
  
  -- Auto update payment_status
  IF COALESCE(NEW.amount_paid, 0) = 0 THEN
    NEW.payment_status = 'unpaid';
  ELSIF NEW.amount_paid >= NEW.total THEN
    NEW.payment_status = 'paid';
    IF NEW.paid_at IS NULL THEN
      NEW.paid_at = NOW();
    END IF;
  ELSIF NEW.amount_paid < NEW.total THEN
    NEW.payment_status = 'partial';
  END IF;
  
  -- Check overdue (only if not paid)
  IF NEW.payment_status != 'paid' AND NEW.due_date < CURRENT_DATE THEN
    NEW.payment_status = 'overdue';
  END IF;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger: Auto calculate amount_due before insert/update
CREATE TRIGGER trigger_calculate_amount_due
  BEFORE INSERT OR UPDATE ON invoices
  FOR EACH ROW EXECUTE FUNCTION calculate_amount_due();

-- Function: Update invoice payment when receipt created/updated/deleted
CREATE OR REPLACE FUNCTION update_invoice_payment()
RETURNS TRIGGER AS $$
DECLARE
  target_invoice_id UUID;
BEGIN
  -- Determine which invoice to update
  IF TG_OP = 'DELETE' THEN
    target_invoice_id = OLD.invoice_id;
  ELSE
    target_invoice_id = NEW.invoice_id;
  END IF;
  
  -- Update total amount paid in invoice
  UPDATE invoices
  SET amount_paid = (
    SELECT COALESCE(SUM(amount_paid), 0)
    FROM receipts
    WHERE invoice_id = target_invoice_id
  )
  WHERE id = target_invoice_id;
  
  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

-- Trigger: Update invoice payment after receipt changes
CREATE TRIGGER trigger_update_invoice_payment
  AFTER INSERT OR UPDATE OR DELETE ON receipts
  FOR EACH ROW EXECUTE FUNCTION update_invoice_payment();

-- Function: Auto create user_profile after signup
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.user_profiles (id, full_name)
  VALUES (NEW.id, COALESCE(NEW.raw_user_meta_data->>'full_name', NEW.email));
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger: Create user_profile on signup
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- Function: Update updated_at timestamp
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Triggers: Auto update updated_at
CREATE TRIGGER update_user_profiles_updated_at BEFORE UPDATE ON user_profiles
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_profiles_updated_at BEFORE UPDATE ON profiles
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_invoices_updated_at BEFORE UPDATE ON invoices
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_invoice_items_updated_at BEFORE UPDATE ON invoice_items
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_tax_settings_updated_at BEFORE UPDATE ON tax_settings
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_reminder_settings_updated_at BEFORE UPDATE ON reminder_settings

-- RLS Policies: invoices

  FOR SELECT USING (
    created_by = auth.uid() OR 
    is_public = true
  );

CREATE POLICY "Users can create invoices" ON invoices
  FOR INSERT WITH CHECK (auth.uid() = created_by);

CREATE POLICY "Users can update own invoices" ON invoices
  FOR UPDATE USING (created_by = auth.uid());

CREATE POLICY "Users can delete own invoices" ON invoices
  FOR DELETE USING (created_by = auth.uid());

-- RLS Policies: invoice_items
CREATE POLICY "Users can view invoice items" ON invoice_items
  FOR SELECT USING (
    EXISTS (
      SELECT 1 FROM invoices 
      WHERE invoices.id = invoice_items.invoice_id 
      AND (invoices.created_by = auth.uid() OR invoices.is_public = true)
    )
  );

CREATE POLICY "Users can manage own invoice items" ON invoice_items
  FOR ALL USING (
    EXISTS (
      SELECT 1 FROM invoices 
      WHERE invoices.id = invoice_items.invoice_id 
      AND invoices.created_by = auth.uid()
    )
  );

-- RLS Policies: receipts
CREATE POLICY "Users can view receipts" ON receipts
  FOR SELECT USING (
    created_by = auth.uid() OR 
    is_public = true OR
    EXISTS (
      SELECT 1 FROM invoices 
      WHERE invoices.id = receipts.invoice_id 
      AND (invoices.created_by = auth.uid() OR invoices.is_public = true)
    )
  );

CREATE POLICY "Users can create receipts" ON receipts
  FOR INSERT WITH CHECK (auth.uid() = created_by);

CREATE POLICY "Users can update own receipts" ON receipts
  FOR UPDATE USING (created_by = auth.uid());

CREATE POLICY "Users can delete own receipts" ON receipts
  FOR DELETE USING (created_by = auth.uid());
